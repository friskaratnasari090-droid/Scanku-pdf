// Global State let pages = []; // Menyimpan objek { id, dataUrl, originalCanvas } let activePageId = null; let videoStream = null; // DOM Elements const views = { home: document.getElementById('view-home'), camera: document.getElementById('view-camera'), editor: document.getElementById('view-editor'), manager: document.getElementById('view-manager') }; const pageCounter = document.getElementById('page-counter'); const loadingOverlay = document.getElementById('loading-overlay'); const loadingText = document.getElementById('loading-text'); // Tombol navigasi & aksi utama document.getElementById('btn-open-camera').addEventListener('click', startCamera); document.getElementById('btn-open-upload').addEventListener('click', () => document.getElementById('file-input').click()); document.getElementById('file-input').addEventListener('change', handleImageUpload); document.getElementById('btn-close-camera').addEventListener('click', stopCamera); document.getElementById('btn-capture').addEventListener('click', captureCameraImage); document.getElementById('btn-add-more').addEventListener('click', () => switchView('home')); document.getElementById('btn-cancel-crop').addEventListener('click', () => switchView(pages.length > 0 ? Camera & File Upload Handler async function startCamera() { switchView('camera'); const video = document.getElementById('camera-stream'); try { videoStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment', width: { ideal: 1920 }, height: { ideal: 1080 } }, audio: false }); video.srcObject = videoStream; } catch (err) { alert('Tidak dapat mengakses kamera perangkat. Pastikan izin kamera diberikan.'); switchView('home'); } } function stopCamera() { if (videoStream) { videoStream.getTracks().forEach(track => track.stop()); videoStream = null; } switchView(pages.length > 0 ? 'manager' : 'home'); } function captureCameraImage() { const video = document.getElementById('camera-stream'); const canvas = document.getElementById('camera-canvas'); canvas.width = video.videoWidth; canvas.height = video.videoHeight; const ctx = canvas.getContext('2d'); ctx.drawImage(video, 0, 0, canvas.width, canvas.height); stopCamera(); loadIntoEditor(canvas); } function handleImageUpload(e) { const files = e.target.files; if (!files.length) return; showLoading('Memuat gambar...'); let processedCount = 0; Array.from(files).forEach(file => { const reader = new FileReader(); reader.onload = function(event) { const img = new Image(); img.onload = function() { const canvas = document.createElement('canvas'); canvas.width = img.width; canvas.height = img.height; const ctx = canvas.getContext('2d'); ctx.drawImage(img, 0, 0); if (processedCount === 0) { loadIntoEditor(canvas); } else { // Jika upload banyak file sekaligus, langsung masukkan ke daftar halaman const pageObj = { id: 'page_' + Date.now() + Math.random(), dataUrl: canvas.toDataURL('image/jpeg', 0.85) }; pages.push(pageObj); updateManagerView(); } processedCount++; if (processedCount === files.length) { hideLoading(); } }; img.src = event.target.result; }; reader.readAsDataURL(file); }); e.target.value = ''; } // 3 & 4. Editor, Crop & Perspective Setup let editorCanvas = document.getElementById('editor-canvas'); let editorCtx = editorCanvas.getContext('2d'); let currentSourceImage = null; let cropPoints = {}; // tl, tr, bl, br function loadIntoEditor(sourceCanvas) { currentSourceImage = sourceCanvas; switchView('editor'); editorCanvas.width = sourceCanvas.width; editorCanvas.height = sourceCanvas.height; editorCtx.drawImage(sourceCanvas, 0, 0); // Inisialisasi sudut crop di 4 sudut gambar dengan margin 10% const w = sourceCanvas.width; const h = sourceCanvas.height; cropPoints = { tl: { x: w * 0.1, y: h * 0.1 }, tr: { x: w * 0.9, y: h * 0.1 }, br: { x: w * 0.9, y: h * 0.9 }, bl: { x: w * 0.1, y: h * 0.9 } }; initCropInteraction(); updateCropUI(); resetFiltersAndSliders(); } function updateCropUI() { const wrapper = document.getElementById('crop-wrapper'); const rect = editorCanvas.getBoundingClientRect(); const scaleX = editorCanvas.width / editorCanvas.offsetWidth; const scaleY = editorCanvas.height / editorCanvas.offsetHeight; // Posisikan elemen HTML handle berdasarkan koordinat canvas asli ['tl', 'tr', 'bl', 'br'].forEach(id => { const handle = document.getElementById('handle-' + id); handle.style.left = (cropPoints[id].x / scaleX) + 'px'; handle.style.top = (cropPoints[id].y / scaleY) + 'px'; }); // Update SVG Polygon const polyPoints = `${cropPoints.tl.x / scaleX},${cropPoints.tl.y / scaleY} ` + `${cropPoints.tr.x / scaleX},${cropPoints.tr.y / scaleY} ` + `${cropPoints.br.x / scaleX},${cropPoints.br.y / scaleY} ` + `${cropPoints.bl.x / scaleX},${cropPoints.bl.y / scaleY}`; document.getElementById('crop-polygon').setAttribute('points', polyPoints); } // Interaksi geser sudut crop (Touch & Mouse Support) function initCropInteraction() { const handles = document.querySelectorAll('.crop-handle'); let activeHandle = null; handles.forEach(handle => { const onStart = (e) => { activeHandle = handle.getAttribute('data-id'); e.preventDefault(); }; handle.addEventListener('mousedown', onStart); handle.addEventListener('touchstart', onStart); }); const onMove = (e) => { if (!activeHandle) return; const wrapper = document.getElementById('crop-wrapper'); const rect = editorCanvas.getBoundingClientRect(); let clientX = e.clientX || (e.touches && e.touches[0].clientX); let clientY = e.clientY || (e.touches && e.touches[0].clientY); let x = (clientX - rect.left) * (editorCanvas.width / editorCanvas.offsetWidth); let y = (clientY - rect.top) * (editorCanvas.height / editorCanvas.offsetHeight); // Batasi dalam area canvas x = Math.max(0, Math.min(editorCanvas.width, x)); y = Math.max(0, Math.min(editorCanvas.height, y)); cropPoints[activeHandle] = { x, y }; updateCropUI(); }; const onEnd = () => { activeHandle = null; }; window.addEventListener('mousemove', onMove); window.addEventListener('touchmove', onMove); window.addEventListener('mouseup', onEnd); window.addEventListener('touchend', onEnd); } // 5 & 6. Filters, Brightness & Contrast Processing const filterBtns = document.querySelectorAll('.filter-btn'); let activeFilter = 'original'; filterBtns.forEach(btn => { btn.addEventListener('click', () => { filterBtns.forEach(b => b.classList.remove('active')); btn.classList.add('active'); activeFilter = btn.getAttribute('data-filter'); redrawEditorCanvas(); }); }); document.getElementById('range-brightness').addEventListener('input', redrawEditorCanvas); document.getElementById('range-contrast').addEventListener('input', redrawEditorCanvas); function resetFiltersAndSliders() { activeFilter = 'original'; filterBtns.forEach(b => b.classList.remove('active')); document.querySelector('[data-filter="original"]').classList.add('active'); document.getElementById('range-brightness').value = 0; document.getElementById('range-contrast').value = 0; redrawEditorCanvas(); } function redrawEditorCanvas() { if (!currentSourceImage) return; editorCtx.clearRect(0, 0, editorCanvas.width, editorCanvas.height); editorCtx.drawImage(currentSourceImage, 0, 0); let imgData = editorCtx.getImageData(0, 0, editorCanvas.width, editorCanvas.height); let data = imgData.data; let brightness = parseInt(document.getElementById('range-brightness').value); let contrast = parseInt(document.getElementById('range-contrast').value); let factor = (259 * (contrast + 255)) / (255 * (259 - contrast)); for (let i = 0; i < data.length; i += 4) { let r = data[i], g = data[i+1], b = data[i+2]; // Filter Effects if (activeFilter === 'grayscale') { let avg = (r + g + b) / 3; r = g = b = avg; } else if (activeFilter === 'magic') { let avg = (r * 0.3 + g * 0.59 + b * 0.11); r = g = b = avg > 140 ? 255 : avg * 0.8; // Peningkatan ketajaman dokumen } else if (activeFilter === 'darken') { let avg = (r + g + b) / 3; r = g = b = avg < 120 ? 0 : 255; } // Brightness & Contrast Adjustment r = factor * (r - 128) + 128 + brightness; g = factor * (g - 128) + 128 + brightness; b = factor * (b - 128) + 128 + brightness; data[i] = Math.min(255, Math.max(0, r)); data[i+1] = Math.min(255, Math.max(0, g)); data[i+2] = Math.min(255, Math.max(0, b)); } editorCtx.putImageData(imgData, 0, 0); } // 4. Auto Perspective Correction & Apply Crop function applyCropAndFilters() { showLoading('Memproses koreksi dokumen...'); setTimeout(() => { // Buat canvas baru hasil crop perspektif sederhana (bounding box / transformasi) const outputCanvas = document.createElement('canvas'); const width = Math.max( Math.hypot(cropPoints.tr.x - cropPoints.tl.x, cropPoints.tr.y - cropPoints.tl.y), Math.hypot(cropPoints.br.x - cropPoints.bl.x, cropPoints.br.y - cropPoints.bl.y) ); const height = Math.max( Math.hypot(cropPoints.bl.x - cropPoints.tl.x, cropPoints.bl.y - cropPoints.tl.y), Math.hypot(cropPoints.br.x - cropPoints.tr.x, cropPoints.br.y - cropPoints.tr.y) ); outputCanvas.width = width; outputCanvas.height = height; const outCtx = outputCanvas.getContext('2d'); // Draw cropped area outCtx.drawImage( editorCanvas, Math.min(cropPoints.tl.x, cropPoints.bl.x), Math.min(cropPoints.tl.y, cropPoints.tr.y), width, height, 0, 0, width, height ); const pageObj = { id: 'page_' + Date.now(), dataUrl: outputCanvas.toDataURL('image/jpeg', 0.9) }; pages.push(pageObj); hideLoading(); updateManagerView(); switchView('manager'); }, 100); } // 7, 8 & 9. Multi-page Manager: Thumbnail, Reorder, Delete function updateManagerView() { const container = document.getElementById('thumbnail-container'); container.innerHTML = ''; pages.forEach((page, index) => { const card = document.createElement('div'); card.className = 'thumb-card'; card.innerHTML = `
Hal ${index + 1}
`; container.appendChild(card); }); if (pages.length > 0) { pageCounter.classList.remove('hidden'); pageCounter.innerText = `${pages.length} Halaman`; } else { switchView('home'); } } window.movePage = function(index, direction) { const target = index + direction; if (target < 0 || target >= pages.length) return; const temp = pages[index]; pages[index] = pages[target]; pages[target] = temp; updateManagerView(); }; window.deletePage = function(index) { if (confirm('Hapus halaman ini?')) { pages.splice(index, 1); updateManagerView(); } }; // 10, 11, 12 & 13. Preview, Generate & Download PDF (jsPDF) function openPreviewModal() { if (pages.length === 0) return alert('Belum ada halaman untuk dibuat PDF.'); modalPdf.classList.remove('hidden'); generatePDFDataUrl((pdfDataUrl) => { const iframe = document.getElementById('pdf-preview-iframe'); const placeholder = document.getElementById('pdf-preview-placeholder'); iframe.src = pdfDataUrl; iframe.classList.remove('hidden'); placeholder.classList.add('hidden'); }); } function generatePDFDataUrl(callback) { const { jsPDF } = window.jspdf; const doc = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' }); pages.forEach((page, index) => { if (index > 0) doc.addPage(); const img = new Image(); img.src = page.dataUrl; // Ukuran standar A4 dalam mm: 210 x 297 const pageWidth = 210; const pageHeight = 297; const imgWidth = pageWidth; const imgHeight = (img.height * pageWidth) / img.width; let renderHeight = imgHeight; let renderWidth = imgWidth; if (renderHeight > pageHeight) { renderHeight = pageHeight; renderWidth = (img.width * pageHeight) / img.height; } doc.addImage(page.dataUrl, 'JPEG', (pageWidth - renderWidth) / 2, (pageHeight - renderHeight) / 2, renderWidth, renderHeight); }); const pdfOutputUrl = doc.output('datauristring'); callback(pdfOutputUrl); } function generateAndDownloadPDF() { const filenameInput = document.getElementById('pdf-filename').value.trim() || 'Dokumen_ScanKu'; const finalFilename = filenameInput.endsWith('.pdf') ? filenameInput : filenameInput + '.pdf'; showLoading('Menyiapkan file PDF...'); setTimeout(() => { const { jsPDF } = window.jspdf; const doc = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' }); pages.forEach((page, index) => { if (index > 0) doc.addPage(); const pageWidth = 210; const pageHeight = 297; const img = new Image(); img.src = page.dataUrl; let imgWidth = pageWidth; let imgHeight = (img.height * pageWidth) / img.width; if (imgHeight > pageHeight) { imgHeight = pageHeight; imgWidth = (img.width * pageHeight) / img.height; } doc.addImage(page.dataUrl, 'JPEG', (pageWidth - imgWidth) / 2, (pageHeight - imgHeight) / 2, imgWidth, imgHeight); }); doc.save(finalFilename); hideLoading(); modalPdf.classList.add('hidden'); }, 500); }
